# SGI IRIX (n32 ABI) on MIPS, for static programs, built with the
# clang-retro LLVM toolchain (https://github.com/clang-retro/llvm). The
# library is PIC, like everything on IRIX, so it links with code built
# with clang's defaults.
[binaries]
c = ['clang', '--target=mips64-sgi-irix', '-mabi=n32', '-march=mips3', '-nostdlib']
ar = 'llvm-ar'
nm = 'llvm-nm'
strip = 'llvm-strip'

[host_machine]
system = 'irix'
cpu_family = 'mips64'
cpu = 'mips3'
endian = 'big'

[properties]
skip_sanity_check = true
